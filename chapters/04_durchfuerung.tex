\section{Implementierung und Durchf\"uhrung der Prototypen}
\label{ch:implementierung}

\subsection{Testumgebung und Rahmenbedingungen}
\label{sec:testumgebung}
\begin{itemize}
    \item Vorstellung der Webanwendung Puma (Projekt- und Kostenverwaltung, SAP-Schnittstelle) im betrieblichen Kontext der JUMO GmbH \& Co. KG.
    \item Bestehender Technologie-Stack (Java, Eclipse, Maven) und Eigenschaften des Firmennetzwerks (Sicherheitsrichtlinien, Proxy).
    \item Spezifikation der Hardware (z.\,B. Entwickler-Laptop) und des Betriebssystems zur Gew\"ahrleistung der Reproduzierbarkeit fehlt noch und muss erg\"anzt werden.
    \item Verwendete Browser (z.\,B. Chrome v120, Edge v120) m\"ussen aufgef\"uhrt werden.
\end{itemize}

\subsection{Beschreibung der Testszenarien}
\label{sec:testszenarien}
\begin{itemize}
    \item Eindeutige Zuweisung, welche Testklassen aus dem Repository welches Testszenario abbilden, fehlt noch.
    \item \textbf{Szenario 1: Login.} Ziel des Tests, Startbedingungen, Durchlauf (Eingabe Username/Passwort, Validierung des Redirects).
    \item \textbf{Szenario 2: Neues Projekt anlegen.} Ziel, Ausf\"ullen des Stammblatts, korrekte Auswahl in Dropdowns (\texttt{select2}), Verifikation der Systemr\"uckmeldung.
    \item \textbf{Szenario 3: Aktivit\"at bearbeiten.} Ziel, Navigation zur bestehenden Aktivit\"at, Modifikation von Werten (z.\,B. Fortschritt, Datum), Speicherung und Verifikation.
\end{itemize}

\subsection{Setup und technische Integration}
\label{sec:setup}
\begin{itemize}
    \item \textbf{Selenium:} Reibungslose Einbindung via \texttt{pom.xml} und Maven; Umgehung des Kompatibilit\"atsproblems mit der im Unternehmen verwendeten, \"alteren JDK-Version.
    \item \textbf{Cypress:} Installationsh\"urden via \texttt{npm} aufgrund der Proxy-Restriktionen des Unternehmens; zwingende Notwendigkeit eines IT-Tickets; Begründung, warum die lokale ZIP-Installation wegen fehlender CI-Skalierbarkeit verworfen wurde.
\end{itemize}

\subsection{Syntax- und Architekturvergleich}
\label{sec:syntaxvergleich}
\begin{itemize}
    \item Architektonische Unterschiede (Selenium als integrierte Bibliothek in der Java/JUnit-Umgebung vs. Cypress als eigenst\"andiges Tooling auf Node.js-Basis) \cite{Spillner2024}.
    \item Vergleich der Ausdrucksst\"arke (nat\"urliche Sprachn\"ahe und Chaining bei Cypress vs. technische Abstraktion bei Selenium).
    \item Wartbarkeit: Notwendigkeit von Hilfsmethoden und Page-Object-Pattern bei Selenium, um Redundanzen im Code zu vermeiden \cite{Liggesmeyer2009}.
    \item Synchronisation: Integrierte Auto-Waits bei Cypress vs. explizite \texttt{wait.until()}-Aufrufe bei dynamischen Elementen in Selenium.
\end{itemize}

\subsection{Code-Gegen\"uberstellung: Projektanlage}
\begin{itemize}
    \item Die nachfolgenden Listings verdeutlichen den reduzierten Code-Overhead bei Cypress.
    \item In Selenium (Listing \ref{lst:selenium_form}) muss explizit auf die Sichtbarkeit oder Klickbarkeit der Elemente gewartet werden, was den Code aufbl\"aht.
    \item Diese Pr\"ufungen sind bei Cypress (Listing \ref{lst:cypress_form}) im \texttt{get()}- und \texttt{type()}-Befehl bereits implizit gekapselt.
\end{itemize}

\begin{lstlisting}[language=Java, caption={Selenium: Projektname und Datum setzen}, label={lst:selenium_form}]
// Selenium: Explizite Waits fuer jedes Element
WebElement projektname = wait.until(
    ExpectedConditions.visibilityOfElementLocated(By.id("project_name"))
);
projektname.clear();
projektname.sendKeys(PROJEKTNAME);

WebElement endDate = wait.until(
    ExpectedConditions.elementToBeClickable(By.id("project_endDate"))
);
endDate.clear();
endDate.sendKeys("17.08.2026");
\end{lstlisting}

\begin{lstlisting}[language=Java, caption={Cypress: Projektname und Datum setzen}, label={lst:cypress_form}]
// Cypress: Implizite Waits durch Chaining
cy.get('#project_name')
  .type('Cypress Testprojekt')

cy.get('#project_endDate')
  .scrollIntoView()
  .clear()
  .type('17.08.2026')
\end{lstlisting}
